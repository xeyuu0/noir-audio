enable_language(RC)
add_library(noir_windows STATIC src/Wasapi.cpp src/Media.cpp src/Broadcast.cpp)
target_include_directories(noir_windows PUBLIC src)
target_link_libraries(noir_windows PUBLIC noir_core Threads::Threads ole32 uuid avrt mfplat mfreadwrite mfuuid propsys)
target_compile_definitions(noir_windows PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_executable(NoirAudio WIN32 src/App.cpp resources/Noir.rc)
target_link_libraries(NoirAudio PRIVATE noir_windows comctl32 gdi32 shell32 comdlg32 uxtheme dwmapi)
set_target_properties(NoirAudio PROPERTIES OUTPUT_NAME "NOIR Audio")
add_executable(NoirWindowsTests tests/WindowsTests.cpp)
target_link_libraries(NoirWindowsTests PRIVATE noir_windows)
add_test(NAME windows_native_services COMMAND NoirWindowsTests)
if(MINGW)
  foreach(t NoirAudio NoirWindowsTests)
    target_link_options(${t} PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--stack,16777216)
  endforeach()
  target_link_options(NoirAudio PRIVATE -municode)
endif()
if(MSVC)
  target_link_options(NoirAudio PRIVATE /MANIFEST:NO)
  target_link_options(NoirWindowsTests PRIVATE /STACK:16777216)
  target_compile_options(noir_windows PRIVATE /utf-8 /EHsc)
  target_compile_options(NoirAudio PRIVATE /utf-8 /EHsc)
  target_compile_options(NoirWindowsTests PRIVATE /utf-8 /EHsc)
endif()
install(TARGETS NoirAudio RUNTIME DESTINATION .)
