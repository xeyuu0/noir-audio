cmake_minimum_required(VERSION 3.24)
project(NoirAudio VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(noir_core INTERFACE)
target_include_directories(noir_core INTERFACE Sources/DSP Sources/AudioEngine ThirdParty/OctopusEAX/src)
target_compile_definitions(noir_core INTERFACE _USE_MATH_DEFINES NOMINMAX)
find_package(Threads REQUIRED)
include(CTest)
add_executable(NoirCoreTests Tests/CoreTests.cpp)
target_include_directories(NoirCoreTests PRIVATE Windows/tests)
target_compile_definitions(NoirCoreTests PRIVATE NOIR_PORTABLE_CORE_TEST)
target_link_libraries(NoirCoreTests PRIVATE noir_core Threads::Threads)
add_test(NAME portable_core COMMAND NoirCoreTests)
add_executable(NoirPortTests Windows/tests/PortTests.cpp)
target_include_directories(NoirPortTests PRIVATE Windows/src)
target_link_libraries(NoirPortTests PRIVATE noir_core Threads::Threads)
add_test(NAME windows_portable_services COMMAND NoirPortTests)
if(WIN32)
  add_subdirectory(Windows)
  if(MINGW)
    foreach(t NoirCoreTests NoirPortTests)
      target_link_options(${t} PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--stack,16777216)
    endforeach()
  endif()
endif()

if(MSVC)
  foreach(t NoirCoreTests NoirPortTests)
    target_compile_options(${t} PRIVATE /utf-8 /EHsc)
    target_link_options(${t} PRIVATE /STACK:16777216)
  endforeach()
endif()
